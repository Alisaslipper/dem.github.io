def play_game():
print("🤖 Игра: Компьютер угадывает ваше число")
print("-" * 40)

try:
n = int(input("Введите нижнюю границу диапазона N: "))
m = int(input("Введите верхнюю границу диапазона M: "))
except ValueError:
print("❌ Ошибка: введите целые числа.")
return

if n >= m:
print("❌ Ошибка: N должно быть строго меньше M.")
return

low, high = n, m
questions = 0

print(f"\n🔹 Загадайте число от {low} до {high}.")
print("🔹 Отвечайте че стно. Можно использовать: да/y/1 или нет/n/0")
print("-" * 40)

while low < high:
mid = (low + high) // 2
questions += 1

is_greater = get_yes_no(f"❓ Вопрос {questions}: Ваше число больше {mid}? (да/нет): ")

if is_greater:
low = mid + 1
else:
high = mid

print("\n" + "=" * 40)
print(f"✅ Я угадал! Ваше число: {low}")
print(f"📊 Использовано вопросов: {questions}")
print(f"📏 Теоретический минимум для этого диапазона: {len(bin(high - low)) - 2} (бинарный поиск)")
print("=" * 40)

if __name__ == "__main__":
play_game()
